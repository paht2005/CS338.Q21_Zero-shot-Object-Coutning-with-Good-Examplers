On the server, run the integrity check:

```bash
python3 - <<'EOF'
import json, os

DATA = "code/source-code/data/FSC147"
FILES = [
    "annotation_FSC147_pos_owlv2.json",
    "annotation_FSC147_neg_owlv2.json",
    "annotation_FSC147_pos_florence2.json",
    "annotation_FSC147_neg_florence2.json",
]

split = json.load(open(f"{DATA}/Train_Test_Val_FSC_147.json"))["val"]
target = set(split)
print(f"Val split: {len(target)} images\n")

for fname in FILES:
    path = f"{DATA}/{fname}"
    if not os.path.exists(path):
        print(f"MISSING: {fname}")
        continue
    d = json.load(open(path))
    missing = target - set(d.keys())
    sample = next(iter(d.values()))
    has_hw = "H" in sample and "W" in sample
    has_boxes = "boxes" in sample
    print(f"{fname}: {len(d)} entries | missing={len(missing)} | H+W={has_hw} | boxes={has_boxes}")
EOF
```

Expected: 4 files, each with 1286 entries, 0 missing, H+W=True, boxes=True.
Florence-2 may produce 0 boxes for some images (empty list is valid — not an error).
