On the server, run the full Phase 3 integrity check:

```bash
python3 - <<'EOF'
import json, os

DATA = "code/source-code/data/FSC147"
FILES = {
    "annotation_FSC147_pos_owlv2.json":         "OWL-v2 pos raw",
    "annotation_FSC147_neg_owlv2.json":         "OWL-v2 neg raw",
    "annotation_FSC147_pos_florence2.json":     "Florence-2 pos raw",
    "annotation_FSC147_neg_florence2.json":     "Florence-2 neg raw",
    "annotation_FSC147_pos_owlv2_prompt.json":   "OWL-v2 pos prompt",
    "annotation_FSC147_neg_owlv2_prompt.json":   "OWL-v2 neg prompt",
    "annotation_FSC147_pos_florence2_prompt.json":"Florence-2 pos prompt",
    "annotation_FSC147_neg_florence2_prompt.json":"Florence-2 neg prompt",
}
split = json.load(open(f"{DATA}/Train_Test_Val_FSC_147.json"))["val"]
target = set(split)
print(f"Val split: {len(target)} images\n")

results = {}
for fname, label in FILES.items():
    path = f"{DATA}/{fname}"
    if not os.path.exists(path):
        print(f"MISSING: {label} ({fname})")
        results[fname] = None
        continue
    d = json.load(open(path))
    total_boxes = sum(len(v.get("boxes", [])) for v in d.values())
    avg_boxes = total_boxes / len(d) if d else 0
    print(f"{label}: {len(d)} entries | avg {avg_boxes:.1f} boxes/img")
    results[fname] = d

# Compare raw vs. prompt for OWL-v2 positive
raw = results.get("annotation_FSC147_pos_owlv2.json")
prom = results.get("annotation_FSC147_pos_owlv2_prompt.json")
if raw and prom:
    raw_avg = sum(len(v.get("boxes", [])) for v in raw.values()) / len(raw)
    prom_avg = sum(len(v.get("boxes", [])) for v in prom.values()) / len(prom)
    print(f"\nOWL-v2 pos: raw avg {raw_avg:.2f} boxes  →  prompt avg {prom_avg:.2f} boxes")
    print("Rich Prompts pipeline: OK" if raw_avg != prom_avg else "WARNING: raw == prompt (queries may not have changed)")
EOF
```

Expected: 8 files, each with 1286 entries. The OWL-v2 raw vs. prompt avg boxes should differ
